#include <tunables/global>

profile esphome_displayeditor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  signal (send) set=(kill,term,int,hup,cont),

  network inet stream,
  network inet6 stream,
  # Hostnames ending in .local are resolved through mDNS/zeroconf.
  network inet dgram,
  network inet6 dgram,

  # The container runs run.sh directly as PID 1 (no S6 Overlay/Bashio).
  /bin/** ix,

  /opt/esphome-displayeditor/bin/python3 ix,
  /opt/esphome-displayeditor/** rix,
  # /app itself must be listable, not just its contents: it is a sys.path
  # entry, so Python scans the directory to find the `backend` package.
  /app/ r,
  /app/** r,
  /run.sh rix,

  /data/ rw,
  /data/** rwk,
  /homeassistant/esphome/ r,
  /homeassistant/esphome/** rwk,

  /etc/** r,
  /proc/** r,
  /sys/devices/system/cpu/** r,
  /tmp/ rw,
  /tmp/** rwk,
}
